ESP32 Filesystem Guide

SPIFFS • LittleFS • FATFS (FFat)

Overview

This guide explains how to create filesystem images, flash them, and use them in ESP32 projects.

Loction

Location of mkspiffs:

C:\Users\\AppData\Local\Arduino15\packages\esp32\tools\mkspiffs\0.2.3\mkspiffs.exe

Location of mklittlefs:

C:\Users\\AppData\Local\Arduino15\packages\esp32\tools\mklittlefs\3.0.0-gnu12-dc7f933\mklittlefs.exe

Project Structure

your_project/
 ├── main.ino
 └── data/
     ├── index.html
     ├── config.json

Step 1: Partition Table

spiffs, data, spiffs, 0x290000, 0x170000,

Offset: 0x290000
Size: 0x170000

Step 2: Create Filesystem

SPIFFS

mkspiffs -c data -p 256 -b 4096 -s 0x170000 spiffs.bin

LittleFS

mklittlefs -c data -p 256 -b 4096 -s 0x170000 littlefs.bin

Step 3: Flash

esptool.py --chip esp32 --port COM5 --baud 921600 write_flash 0x290000 spiffs.bin

Example Code (SPIFFS)

#include <SPIFFS.h>
void setup() {
  Serial.begin(115200); 
  if (!SPIFFS.begin(true))
  {
    Serial.println("SPIFFS Mount Failed"); 
    return; 
  } 
  File file = SPIFFS.open("/config.json", "r");
  if (file)
  {
    Serial.println(file.readString());
    file.close();
  }
}
void loop() {}

Example Code (LittleFS)

#include <LittleFS.h>

void setup()
{
   Serial.begin(115200);
   if (!LittleFS.begin(true))
   {
     Serial.println("LittleFS Mount Failed");
     return;

    {
       Serial.println(file.readString());
       file.close();
    }
}
void loop() {}

Example Code (FFat)

#include <FFat.h>

void setup() {
  Serial.begin(115200);

  if (!FFat.begin(true)) {
    Serial.println("FFat Mount Failed");
    return;
  }

  // Write file
  File file = FFat.open("/test.txt", FILE_WRITE);
  file.println("Hello from FATFS!");
  file.close();

  // Read file
  file = FFat.open("/test.txt");
  while (file.available()) {
    Serial.write(file.read());
  }
  file.close();
}

void loop() {}
